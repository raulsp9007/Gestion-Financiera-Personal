🔗 Sincronizar con Google Sheets

Pasos de configuración (se hace una sola vez):
  1. Abre tu Google Spreadsheet ↗
  2. Ve a Extensiones → Apps Script
  3. Borra todo el código existente y pega el siguiente:
// Zelle 2026 - Sync Script const SS_ID = '1WDH6KQE2RnMIi7BxFu28ptTcti8IeUu8'; const TAB = 'ZelleApp'; function doGet(e) { const d = e.parameter.data ? JSON.parse(decodeURIComponent(e.parameter.data)) : { action: 'read' }; return respond(handle(d)); } function handle(d) { const ss = SpreadsheetApp.openById(SS_ID); let sh = ss.getSheetByName(TAB); if (!sh) { sh = ss.insertSheet(TAB); sh.appendRow(['id','date','amount','description','type','category','notes']); } if (d.action === 'read') { if (sh.getLastRow() <= 1) return []; const [h, ...rows] = sh.getDataRange().getValues(); return rows.map(r => ({ id: +r[0], date: r[1], amount: +r[2], description: r[3], type: r[4], category: r[5], notes: r[6] })); } if (d.action === 'clear') { if (sh.getLastRow() > 1) sh.deleteRows(2, sh.getLastRow() - 1); return { ok: true }; } if (d.action === 'add_batch') { const vals = d.batch.map(t => [t.id, t.date, t.amount, t.description||'', t.type, t.category, t.notes||''] ); sh.getRange(sh.getLastRow() + 1, 1, vals.length, 7).setValues(vals); return { ok: true }; } if (d.action === 'add') { const t = d.tx; sh.appendRow([t.id, t.date, t.amount, t.description||'', t.type, t.category, t.notes||'']); return { ok: true }; } if (d.action === 'update') { const vals = sh.getDataRange().getValues(); for (let i = 1; i < vals.length; i++) { if (String(vals[i][0]) === String(d.tx.id)) { const t = d.tx; sh.getRange(i + 1, 1, 1, 7).setValues( [[t.id, t.date, t.amount, t.description||'', t.type, t.category, t.notes||'']] ); return { ok: true }; } } return { error: 'not found' }; } if (d.action === 'delete') { const vals = sh.getDataRange().getValues(); for (let i = 1; i < vals.length; i++) { if (String(vals[i][0]) === String(d.id)) { sh.deleteRow(i + 1); return { ok: true }; } } return { error: 'not found' }; } return { error: 'unknown action' }; } function respond(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
  1. Guarda el proyecto (Ctrl+S / ⌘S)
